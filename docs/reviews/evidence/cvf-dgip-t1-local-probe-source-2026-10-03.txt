def put(root, rel, value):
    path = root / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(value if isinstance(value,str) else json.dumps(value,indent=2)+'\n',encoding='utf-8')

def invoke(root, *args, runner=RUNNER):
    done = subprocess.run([sys.executable,'-B',str(runner),*args,'--project-root',str(root)],env=ENV,cwd=root,text=True,capture_output=True)
    return {'exitCode':done.returncode,'stdout':done.stdout,'stderr':done.stderr}

def case(name, mutate, phase='pr-ci', packet=None, pinned=False, expected='REFUSE'):
    root=OUT/name
    shutil.copytree(BASE,root)
    mutate(root)
    args=['run','--phase',phase,'--json','--no-receipt']
    if packet: args += ['--packet',packet]
    if pinned: args += pins
    done=invoke(root,*args,runner=root/'scripts/cvf_gates/cvf_downstream_gate_runner.py')
    actual='PASS' if done['exitCode']==0 else 'REFUSE'
    results.append({'id':name,'expected':expected,'actual':actual,'oraclePass':actual==expected,**done})
    return root
